cmake_minimum_required(VERSION 3.20)
project(AIPrisonBreak VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(raylib)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
add_library(prison_world src/world.cpp)
target_include_directories(prison_world PUBLIC src)
add_executable(AIPrisonBreak src/main.cpp src/ai.cpp src/agent.cpp)
target_link_libraries(AIPrisonBreak PRIVATE raylib prison_world nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(AIPrisonBreak PRIVATE winhttp)
  set_target_properties(AIPrisonBreak PROPERTIES WIN32_EXECUTABLE TRUE)
  if(MSVC)
    # Native GUI subsystem while retaining the standard int main() entry point.
    target_link_options(AIPrisonBreak PRIVATE "/ENTRY:mainCRTStartup")
  endif()
endif()
enable_testing()
add_executable(world_test tests/world_test.cpp)
target_link_libraries(world_test PRIVATE prison_world)
add_test(NAME world_test COMMAND world_test)

add_executable(agent_test tests/agent_test.cpp src/agent.cpp)
target_include_directories(agent_test PRIVATE src)
target_link_libraries(agent_test PRIVATE prison_world)
add_test(NAME agent_test COMMAND agent_test)

add_executable(presentation_test tests/presentation_test.cpp)
target_include_directories(presentation_test PRIVATE src)
add_test(NAME presentation_test COMMAND presentation_test)

add_executable(ai_test tests/ai_test.cpp src/ai.cpp)
target_include_directories(ai_test PRIVATE src)
target_link_libraries(ai_test PRIVATE prison_world nlohmann_json::nlohmann_json)
if(WIN32)
    target_link_libraries(ai_test PRIVATE winhttp)
endif()
add_test(NAME ai_test COMMAND ai_test)
